# FortiGate Wireguard Configuration for Multi-Organization Zero Trust Architecture

# ==========================================
# ER1 ORGANIZATION CONFIGURATION
# ==========================================

# 1. Create Wireguard interface for ER1
config system interface
    edit "wg-to-er1"
        set type tunnel
        set tunnel-type wireguard
        set interface "wan1"
        set ip 192.168.100.1/30
        set allowaccess ping ssh
        set mtu-override enable
        set mtu 1420
    next
end

# 2. Configure Wireguard peer for ER1
config vpn wireguard
    edit "wg-to-er1"
        set interface "wg-to-er1"
        set port 51820
        set private-key "FORTIGATE_PRIVATE_KEY_HERE"
        config peer
            edit 1
                set public-key "ER1_ROUTER_PUBLIC_KEY_HERE"
                set endpoint "ER1_ROUTER_PUBLIC_IP:51820"
                set allowed-ips 10.18.11.0/24
                set persistent-keepalive 25
            next
        end
    next
end

# 3. Static route for ER1 private network
config router static
    edit 1
        set dst 10.18.11.0/24
        set gateway 192.168.100.2
        set device "wg-to-er1"
        set status enable
    next
end

# ==========================================
# ZERO TRUST FIREWALL POLICIES
# ==========================================

# 10. Zero Trust policy: Organizations to Internet (monitored)
config firewall policy
    edit 10
        set name "ER1-to-Internet"
        set srcintf "wg-to-er1"
        set dstintf "wan1"
        set srcaddr "ER1-Network"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
        set logtraffic all
        set comments "Zero Trust: ER1 internet access via FortiGate"
    next
    edit 11
        set name "ER2-to-Internet"
        set srcintf "wg-to-er2"
        set dstintf "wan1"
        set srcaddr "ER2-Network"
        set dstaddr "all"

end

# ==========================================
# NETWORK ADDRESS OBJECTS
# ==========================================

# 13. Address objects for organization networks
config firewall address
    edit "ER1-Network"
        set subnet 10.18.11.0/24
        set comment "ER1 Organization Private Network"
    next
    edit "ER2-Network"
        set subnet 10.18.21.0/24
        set comment "ER2 Organization Private Network"
    next
    edit "ER3-Network"
        set subnet 10.18.31.0/24
        set comment "ER3 Organization Private Network"
    next
    edit "All-Organizations"
        set type group
        set member "ER1-Network" "ER2-Network" "ER3-Network"
        set comment "All organization networks for Zero Trust monitoring"
    next
end

# DEPLOYMENT NOTES

# Key Replacements Required:
# - FORTIGATE_PRIVATE_KEY_HERE: FortiGate's Wireguard private key
# - ER1_ROUTER_PUBLIC_KEY_HERE: ER1 router VM's public key  
# - ER1_ROUTER_PUBLIC_IP: ER1 router VM's public IP
# - ER2_ROUTER_PUBLIC_KEY_HERE: ER2 router VM's public key
# - ER2_ROUTER_PUBLIC_IP: ER2 router VM's public IP  
# - ER3_ROUTER_PUBLIC_KEY_HERE: ER3 router VM's public key
# - ER3_ROUTER_PUBLIC_IP: ER3 router VM's public IP

# Zero Trust Architecture Features:
# ✅ Multi-organization support (ER1, ER2, ER3)
# ✅ Isolated tunnel networks (192.168.100.x, 192.168.101.x, 192.168.102.x)
# ✅ Centralized internet access monitoring
# ✅ FortiGate management access to all private instances
# ✅ Inter-organization isolation (organizations cannot access each other)
# ✅ Complete traffic logging and monitoring
# ✅ Different Wireguard ports for organization separation

# Network Architecture Summary:
# Organization | Private Network | Tunnel Network    | FortiGate Port
# ER1         | 10.18.11.0/24   | 192.168.100.0/30 | 51820
# ER2         | 10.18.21.0/24   | 192.168.101.0/30 | 51821  
# ER3         | 10.18.31.0/24   | 192.168.102.0/30 | 51822

# Security Policy Overview:
# 1. Organizations can access internet (monitored via FortiGate NAT)
# 2. FortiGate can manage all private instances (SSH/ping)
# 3. Organizations are isolated from each other (deny policies)
# 4. All traffic is logged for security monitoring
# 5. No direct router-to-private access (Zero Trust feature)

# Production Deployment:
# - Adjust policy numbers to avoid conflicts with existing policies
# - Consider implementing time-based access restrictions
# - Add geo-blocking policies if required
# - Configure IPS/antivirus scanning for additional security
# - Set up logging to SIEM for comprehensive monitoring
